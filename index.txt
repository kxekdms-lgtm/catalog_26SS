<script type="text/javascript">
function loadApp() {

    $('#canvas').fadeIn(1000);

    var flipbook = $('.magazine');
    var cHeight = $(document).height() - $('body').offset().top;
    cHeight = cHeight - 350;
    var cWidth = cHeight*1028/700;

    $('.magazine').turn({
        width: cWidth,
        height: cHeight,
        elevation: 50,
        gradients: true,
        autoCenter: true
    });
}

<script>
$(document).ready(function() {

    loadApp();

    // 다음 페이지
    $('.next').click(function() {
        $('.magazine').turn('next');
    });

    // 이전 페이지
    $('.prev').click(function() {
        $('.magazine').turn('previous');
    });

    // 첫 페이지
    $('.first').click(function() {
        $('.magazine').turn('page', 1);
    });

    // 마지막 페이지
    $('.last').click(function() {
        $('.magazine').turn('page', $('.magazine').turn('pages'));
    });

    // 숫자 입력 후 이동

$('.gotopage').click(function() {

    var page = parseInt($('#topage').val());
    var totalPages = $('.magazine').turn('pages');

    if (!isNaN(page) && page >= 1 && page <= totalPages) {
        $('.magazine').turn('page', page);
    }

});

var isZoomed = false;
var zoomScale = 1.8;

function zoomIn(){
    $('.magazine').css({
        'transform': 'scale(' + zoomScale + ')',
        'transform-origin': 'center center',
        'transition': 'transform 0.3s ease',
        'cursor': 'zoom-out'
    });
    isZoomed = true;
}

function zoomOut(){
    $('.magazine').css({
        'transform': 'scale(1)',
        'cursor': 'default'
    });
    isZoomed = false;
}

function toggleZoom(){
    if(isZoomed){
        zoomOut();
    } else {
        zoomIn();
    }
}

$('.btn_all').click(function(e){
    e.stopPropagation();
    toggleZoom();
});

$('.magazine').click(function(){
    toggleZoom();
});

});

</script>
